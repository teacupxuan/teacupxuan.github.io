北医三院牙体牙髓号总是很难挂上，我之前在另外一个博客上学到了如何进行检测余号并发送，可惜那个博主把文章删了。网上有很多人找我要教程，于是我自己总结一下进行分享。

### 1. 对手机进行抓包
我用的是Charles进行抓包，具体用Charles怎么抓包可以参考网上已有的教程（以后有时候我补充完整）。
抓包后的效果如下：
![charles](/images/charles.png "charles")

不难发现：
https://mp.mhealth100.com/gateway/registration/appointment/scheduleNew/find?branchCode=100238001&deptId=1100101&deptName=%25E5%258F%25A3%25E8%2585%2594%25E7%25A7%2591%25E9%2597%25A8%25E8%25AF%258A&deptType=&startDate=2026-01-10&endDate=2026-01-16&ajaxConfig=true 
就是我们需要的url，然后可以在Charles中右键复制对应的cURL Request就可以知道发送请求所带着的token。注意这个token（即curl中的Cookie）会过期，所以需要定时更换。复制自己的Cookie，后续的脚本需要。

```json
curl -H "Host: mp.mhealth100.com" -H "Cookie:复制这里面的内容" -H "accept: application/json" -H "sec-fetch-site: same-origin" -H "sec-fetch-dest: empty" -H "accept-language: zh-CN,zh-Hans;q=0.9" -H "sec-fetch-mode: cors" -H "user-agent: Mozilla/5.0 (iPhone; CPU iPhone OS 18_6_2 like Mac OS X) AppleWebKit/605.1.15 (KHTML, like Gecko) Mobile/15E148 MicroMessenger/8.0.67(0x1800432d) NetType/WIFI Language/zh_CN" -H "referer: https://mp.mhealth100.com/patient/registration/" --compressed "https://mp.mhealth100.com/gateway/registration/appointment/scheduleNew/find?branchCode=100238001&deptId=1100101&deptName=%25E5%258F%25A3%25E8%2585%2594%25E7%25A7%2591%25E9%2597%25A8%25E8%25AF%258A&deptType=&startDate=2026-01-10&endDate=2026-01-16&ajaxConfig=true"
```

### 2. 手机推送实现

在手机下载Bark软件，下载后复制里面对应的：
Bark Key类似于： https://api.day.app/xxxxx
### 3. 余号检测脚本

电脑需要有安装node，如果没有node则自行进行安装。
安装后，修改以下内容则可以运行成功：
- Cookie
- BARK_URL
- startDate
- endDate

```javascript
/**

 * Doctor Filter Script (Auto Watch)

 * 每 10 秒检测一次

 * 发现牙体牙髓有号 → Bark 推送

 */

  

const fs = require("fs");

  

// ======================= 配置 =======================

const CHECK_INTERVAL = 10 * 1000; // 10 秒

  

const URL =

    "https://mp.mhealth100.com/gateway/registration/appointment/scheduleNew/find" +

    "?branchCode=100238001" +

    "&deptId=1100101" +

    "&deptName=%25E5%258F%25A3%25E8%2585%2594%25E7%25A7%2591%25E9%2597%25A8%25E8%25AF%258A" +

    "&deptType=" +

    "&startDate=2026-01-09" +

    "&endDate=2026-01-16" +

    "&ajaxConfig=true";

  

// ⚠️ Cookie 必须是最新的

const COOKIE =

    "";

  

// Bark Key（下载Bark后复制的）

const BARK_URL = "";

  

// ======================= 状态 =======================

let lastHasAvailable = false; // 防止重复推送

  

// ======================= 请求 =======================

async function fetchDoctors() {

    const res = await fetch(URL, {

        headers: {

            "Accept": "application/json",

            "Referer": "https://mp.mhealth100.com/patient/registration/",

            "User-Agent":

                "Mozilla/5.0 (iPhone; CPU iPhone OS 18_6_2 like Mac OS X) " +

                "AppleWebKit/605.1.15 (KHTML, like Gecko) " +

                "Mobile/15E148 MicroMessenger/8.0.67 NetType/WIFI Language/zh_CN",

            "Cookie": COOKIE

        }

    });

  

    if (!res.ok) {

        throw new Error(`HTTP ${res.status}`);

    }

  

    return res.json();

}

  

// ======================= 过滤 =======================

function findAvailableDoctors(json) {

    const keyword = "牙体牙髓";

    const doctors = json?.data?.regInfos ?? [];

  

    const result = [];

  

    doctors.forEach(doc => {

        if (!doc.doctorTitle?.includes(keyword)) return;

  

        const available = (doc.scheduleInfos || []).filter(

            s => s.regLeaveCount > 0

        );

  

        if (available.length > 0) {

            result.push({

                doctorName: doc.doctorName,

                doctorTitle: doc.doctorTitle,

                schedules: available

            });

        }

    });

  

    return result;

}

  

// ======================= Bark 推送 =======================

async function sendBark(doctors) {

    const title = "🦷 牙体牙髓有号了！";

    const body = doctors.map(d =>

        `${d.doctorName}（${d.doctorTitle}）\n` +

        d.schedules.map(

            s => `- ${s.regDate} ${s.shiftName} 剩余${s.regLeaveCount}号`

        ).join("\n")

    ).join("\n\n");

  

    const url = `${BARK_URL}/${encodeURIComponent(title)}/${encodeURIComponent(body)}`;

  

    await fetch(url);

}

  

// ======================= 主循环 =======================

async function checkOnce() {

    try {

        const json = await fetchDoctors();

        const availableDoctors = findAvailableDoctors(json);

  

        const hasAvailable = availableDoctors.length > 0;

        const now = new Date().toLocaleTimeString();

  

        if (hasAvailable) {

            console.log(`🚨 [${now}] 发现有号医生！`);

  

            if (!lastHasAvailable) {

                await sendBark(availableDoctors);

                console.log("📲 Bark 推送已发送");

            }

        } else {

            console.log(`⏳ [${now}] 暂无放号`);

            await sendBark(availableDoctors);

            console.log("📲 Bark 推送已发送");

        }

  

        lastHasAvailable = hasAvailable;

  

    } catch (err) {

        console.error("❌ 检测失败:", err.message);

    }

}

  

// ======================= 启动 =======================

console.log("🟢 牙体牙髓抢号监控已启动（每 10 秒检测一次）");

checkOnce();

setInterval(checkOnce, CHECK_INTERVAL);
```
